-- ============================================== -- InspiRED - contas + ranking online (Supabase) -- Rode este arquivo UMA VEZ no SQL Editor do Supabase. -- ============================================== create extension if not exists pgcrypto; create table if not exists public.inspired_students ( id uuid primary key default gen_random_uuid(), display_name text not null, school text not null, grade text not null, name_key text not null, school_key text not null, pin_hash text not null, session_token uuid not null default gen_random_uuid(), xp integer not null default 0 check (xp >= 0), questions integer not null default 0 check (questions >= 0), streak integer not null default 0 check (streak >= 0), progress integer not null default 0 check (progress between 0 and 100), created_at timestamptz not null default now(), updated_at timestamptz not null default now(), unique (name_key, school_key) ); alter table public.inspired_students enable row level security; -- Ninguém acessa a tabela diretamente pelo navegador. revoke all on table public.inspired_students from anon, authenticated; -- ============================================== -- CRIAR CONTA -- ============================================== create or replace function public.inspired_register( p_name text, p_school text, p_grade text, p_pin text ) returns table( ok boolean, reason text, student_id uuid, session_token uuid, display_name text, school text, grade text, xp integer, questions integer, streak integer, progress integer ) language plpgsql security definer set search_path = public, extensions as $$ declare v_name_key text := lower(trim(regexp_replace(coalesce(p_name,''), '[[:space:]]+', ' ', 'g'))); v_school_key text := lower(trim(regexp_replace(coalesce(p_school,''), '[[:space:]]+', ' ', 'g'))); v_row public.inspired_students%rowtype; begin if char_length(trim(coalesce(p_name,''))) < 2 or char_length(trim(coalesce(p_school,''))) < 2 then return query select false,'INVALID_DATA',null::uuid,null::uuid,null::text,null::text,null::text,0,0,0,0; return; end if; if p_pin !~ '^[0-9]{4,8}$' then return query select false,'PIN_INVALID',null::uuid,null::uuid,null::text,null::text,null::text,0,0,0,0; return; end if; select s.* into v_row from public.inspired_students s where s.name_key=v_name_key and s.school_key=v_school_key limit 1; if found then return query select false,'ACCOUNT_EXISTS',v_row.id,null::uuid,v_row.display_name,v_row.school,v_row.grade,v_row.xp,v_row.questions,v_row.streak,v_row.progress; return; end if; insert into public.inspired_students( display_name,school,grade,name_key,school_key,pin_hash,session_token ) values ( trim(p_name),trim(p_school),trim(p_grade),v_name_key,v_school_key, crypt(p_pin,gen_salt('bf')),gen_random_uuid() ) returning * into v_row; return query select true,'CREATED',v_row.id,v_row.session_token,v_row.display_name,v_row.school,v_row.grade,v_row.xp,v_row.questions,v_row.streak,v_row.progress; end; $$; -- ============================================== -- ENTRAR EM CONTA EXISTENTE -- ============================================== create or replace function public.inspired_login( p_name text, p_school text, p_pin text ) returns table( ok boolean, reason text, student_id uuid, session_token uuid, display_name text, school text, grade text, xp integer, questions integer, streak integer, progress integer ) language plpgsql security definer set search_path = public, extensions as $$ declare v_name_key text := lower(trim(regexp_replace(coalesce(p_name,''), '[[:space:]]+', ' ', 'g'))); v_school_key text := lower(trim(regexp_replace(coalesce(p_school,''), '[[:space:]]+', ' ', 'g'))); v_row public.inspired_students%rowtype; v_token uuid; begin select s.* into v_row from public.inspired_students s where s.name_key=v_name_key and s.school_key=v_school_key limit 1; if not found then return query select false,'NOT_FOUND',null::uuid,null::uuid,null::text,null::text,null::text,0,0,0,0; return; end if; if v_row.pin_hash <> crypt(coalesce(p_pin,''),v_row.pin_hash) then return query select false,'PIN_WRONG',v_row.id,null::uuid,v_row.display_name,v_row.school,v_row.grade,v_row.xp,v_row.questions,v_row.streak,v_row.progress; return; end if; v_token:=gen_random_uuid(); update public.inspired_students s set session_token=v_token,updated_at=now() where s.id=v_row.id returning s.* into v_row; return query select true,'LOGGED_IN',v_row.id,v_row.session_token,v_row.display_name,v_row.school,v_row.grade,v_row.xp,v_row.questions,v_row.streak,v_row.progress; end; $$; -- ============================================== -- SALVAR PROGRESSO -- ============================================== create or replace function public.inspired_save_progress( p_student_id uuid, p_session_token uuid, p_xp integer, p_questions integer, p_streak integer, p_progress integer ) returns boolean language plpgsql security definer set search_path = public as $$ declare v_count integer; begin update public.inspired_students s set xp=greatest(s.xp,greatest(coalesce(p_xp,0),0)), questions=greatest(s.questions,greatest(coalesce(p_questions,0),0)), streak=greatest(s.streak,greatest(coalesce(p_streak,0),0)), progress=greatest(s.progress,least(100,greatest(coalesce(p_progress,0),0))), updated_at=now() where s.id=p_student_id and s.session_token=p_session_token; get diagnostics v_count = row_count; return v_count=1; end; $$; -- ============================================== -- RANKING -- Retorna nome abreviado: "Kaio R." em vez do nome completo. -- ============================================== create or replace function public.inspired_leaderboard(p_limit integer default 25) returns table( pos bigint, student_id uuid, public_name text, school text, xp integer, questions integer ) language sql security definer stable set search_path = public as $$ select row_number() over(order by s.xp desc,s.questions desc,s.updated_at asc) as pos, s.id as student_id, case when array_length(regexp_split_to_array(trim(s.display_name),'[[:space:]]+'),1) > 1 then split_part(trim(s.display_name),' ',1) || ' ' || upper(left(split_part(trim(s.display_name),' ',2),1)) || '.' else trim(s.display_name) end as public_name, s.school, s.xp, s.questions from public.inspired_students s order by s.xp desc,s.questions desc,s.updated_at asc limit greatest(1,least(coalesce(p_limit,25),100)); $$; -- Só as funções acima ficam disponíveis para o site. grant execute on function public.inspired_register(text,text,text,text) to anon, authenticated; grant execute on function public.inspired_login(text,text,text) to anon, authenticated; grant execute on function public.inspired_save_progress(uuid,uuid,integer,integer,integer,integer) to anon, authenticated; grant execute on function public.inspired_leaderboard(integer) to anon, authenticated;
